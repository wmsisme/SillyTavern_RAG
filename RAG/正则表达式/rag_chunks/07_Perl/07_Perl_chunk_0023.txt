Building and Using Regex Objects

my $HostnameRegex =qr/[-a-z0-9]+(?:\.[-a-z0-9]+)*\.(?:comledulinfo)/i;

my SHttpUrl =qr{

http:// SHostnameRegex \b

(?:

/[-a-z0-9-:\@&?=+,.!/~*'8\$]*

# 可能出现的path

# 不容许以[.,?!] 结尾

my SHostnameRegex = qr{

# 一个或多个点号分隔部分

(?:[a-z0-9]\.I[a-z0-9][-a-z0-9](0,61}[a-z0-9]\.)*

# 后缀

(?:comledulgovlint Imil|net lorglbizlinfol…laerol[a-z][a-z]) }xi;
